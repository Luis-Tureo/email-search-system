CREATE TABLE public.comunas ( id integer NOT NULL DEFAULT nextval('comunas_id_seq'::regclass), name text NOT NULL, region_id integer NOT NULL, CONSTRAINT comunas_pkey PRIMARY KEY (id), CONSTRAINT fk_comunas_region FOREIGN KEY (region_id) REFERENCES public.regions(id) ); CREATE TABLE public.institution_files ( id integer NOT NULL DEFAULT nextval('institution_files_id_seq'::regclass), institution_id integer NOT NULL, file_name text NOT NULL, file_path text NOT NULL, uploaded_at timestamp without time zone DEFAULT now(), CONSTRAINT institution_files_pkey PRIMARY KEY (id), CONSTRAINT fk_files_institution FOREIGN KEY (institution_id) REFERENCES public.institutions(id) ); CREATE TABLE public.institutions ( id integer NOT NULL DEFAULT nextval('institutions_id_seq'::regclass), institution_name text NOT NULL, rut text, email text NOT NULL, phone text, extension text, mobile text, zone_id integer NOT NULL, region_id integer NOT NULL, comuna_id integer NOT NULL, submission_method_id integer NOT NULL, validation_status_id integer NOT NULL, observation text, verified boolean DEFAULT false, active boolean DEFAULT true, created_at timestamp without time zone DEFAULT now(), updated_at timestamp without time zone, modified_by text, CONSTRAINT institutions_pkey PRIMARY KEY (id), CONSTRAINT fk_institutions_zone FOREIGN KEY (zone_id) REFERENCES public.zones(id), CONSTRAINT fk_institutions_region FOREIGN KEY (region_id) REFERENCES public.regions(id), CONSTRAINT fk_institutions_comuna FOREIGN KEY (comuna_id) REFERENCES public.comunas(id), CONSTRAINT fk_institutions_submission_method FOREIGN KEY (submission_method_id) REFERENCES public.submission_methods(id), CONSTRAINT fk_institutions_validation_status FOREIGN KEY (validation_status_id) REFERENCES public.validation_statuses(id) ); CREATE TABLE public.regions ( id integer NOT NULL DEFAULT nextval('regions_id_seq'::regclass), name text NOT NULL, zone_id integer NOT NULL, CONSTRAINT regions_pkey PRIMARY KEY (id), CONSTRAINT fk_regions_zone FOREIGN KEY (zone_id) REFERENCES public.zones(id) ); CREATE TABLE public.submission_methods ( id integer NOT NULL DEFAULT nextval('submission_methods_id_seq'::regclass), name text NOT NULL UNIQUE, CONSTRAINT submission_methods_pkey PRIMARY KEY (id) ); CREATE TABLE public.system_credentials ( id integer NOT NULL DEFAULT nextval('system_credentials_id_seq'::regclass), password_hash text NOT NULL, is_admin boolean DEFAULT true, active boolean DEFAULT true, created_at timestamp without time zone DEFAULT now(), updated_at timestamp without time zone, CONSTRAINT system_credentials_pkey PRIMARY KEY (id) ); CREATE TABLE public.validation_statuses ( id integer NOT NULL DEFAULT nextval('validation_statuses_id_seq'::regclass), name text NOT NULL UNIQUE, CONSTRAINT validation_statuses_pkey PRIMARY KEY (id) ); CREATE TABLE public.zones ( id integer NOT NULL DEFAULT nextval('zones_id_seq'::regclass), name text NOT NULL UNIQUE, CONSTRAINT zones_pkey PRIMARY KEY (id) );
